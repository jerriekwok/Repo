**UpdateGameState() - 状态更新**
```csharp
private void UpdateGameState()
{
    switch (currentGameState)
    {
        case GameState.WaitingToStart:
            // 等待玩家交互开始游戏
            break;
        case GameState.CountdownToStart:
            countdownToStartTimer -= Time.deltaTime;
            if (countdownToStartTimer <= 0f)
            {
                currentGameState = GameState.GamePlaying;
                gamePlayingTimer = gamePlayingTimerMax;
                OnGameStateChanged?.Invoke(this, EventArgs.Empty);
            }
            OnCountdownToStartChanged?.Invoke(this, EventArgs.Empty);
            break;
        case GameState.GamePlaying:
            gamePlayingTimer -= Time.deltaTime;
            if (gamePlayingTimer <= 0f)
            {
                currentGameState = GameState.GameOver;
                OnGameStateChanged?.Invoke(this, EventArgs.Empty);
            }
            OnGamePlayingTimerChanged?.Invoke(this, EventArgs.Empty);
            break;
        case GameState.GameOver:
            // 游戏结束逻辑
            break;
    }
}
```

**HandleMovement() - 移动处理**
```csharp
private void HandleMovement()
{
    // 获取移动输入
    inputVector = GameInput.Instance.GetMovementVectorNormalized();
    
    // 计算移动方向和旋转
    Vector3 moveDirection = new Vector3(inputVector.x, 0f, inputVector.y);
    
    // 移动
    transform.position += moveDirection * moveSpeed * Time.deltaTime;
    
    // 旋转
    if (moveDirection != Vector3.zero)
    {
        transform.forward = Vector3.Slerp(transform.forward, moveDirection, Time.deltaTime * rotateSpeed);
    }
}
```

**HandleInteractions() - 交互处理**
```csharp
private void HandleInteractions()
{
    // 射线检测柜台
    Vector3 raycastOffset = new Vector3(0f, 0.5f, 0f);
    RaycastHit raycastHit;
    if (Physics.Raycast(transform.position + raycastOffset, transform.forward, out raycastHit, interactDistance, countersLayerMask))
    {
        // 检测到柜台
        if (raycastHit.transform.TryGetComponent(out BaseCounter baseCounter))
        {
            if (selectedCounter != baseCounter)
            {
                // 更新选中的柜台
                selectedCounter = baseCounter;
                SetSelectedCounter(baseCounter);
            }
        }
    }
    else
    {
        // 未检测到柜台
        if (selectedCounter != null)
        {
            selectedCounter = null;
            SetSelectedCounter(null);
        }
    }
}
```

### 2.4 厨房物品系统 (KitchenObject)

#### 2.4.1 类设计
```csharp
public class KitchenObject : MonoBehaviour
{
    // 物品数据
    [SerializeField] private KitchenObjectSO kitchenObjectSO;
    
    // 物品父对象
    private IKitchenObjectParent kitchenObjectParent;
    
    // 获取物品数据
    public KitchenObjectSO GetKitchenObjectSO()
    {
        return kitchenObjectSO;
    }
    
    // 设置物品父对象
    public void SetKitchenObjectParent(IKitchenObjectParent kitchenObjectParent)
    {
        // 清除之前的父对象
        if (this.kitchenObjectParent != null)
        {
            this.kitchenObjectParent.ClearKitchenObject();
        }
        
        // 设置新的父对象
        this.kitchenObjectParent = kitchenObjectParent;
        
        if (kitchenObjectParent.HasKitchenObject())
        {
            Debug.LogError("KitchenObjectParent already has a KitchenObject!");
        }
        
        kitchenObjectParent.SetKitchenObject(this);
        
        // 设置物品位置
        transform.parent = kitchenObjectParent.GetKitchenObjectFollowTransform();
        transform.localPosition = Vector3.zero;
    }
}
```

**TryAddIngredient() - 添加食材**
```csharp
public bool TryAddIngredient(KitchenObjectSO kitchenObjectSO)
{
    // 检查是否为有效食材
    if (!validKitchenObjectSOList.Contains(kitchenObjectSO))
    {
        return false;
    }
    
    // 检查是否已经添加过该食材
    if (kitchenObjectSOList.Contains(kitchenObjectSO))
    {
        return false;
    }
    else
    {
        // 添加食材
        kitchenObjectSOList.Add(kitchenObjectSO);
        OnIngredientAdded?.Invoke(this, new OnIngredientAddedEventArgs { kitchenObjectSO = kitchenObjectSO });
        return true;
    }
}
```
